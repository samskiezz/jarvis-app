# WC2026 prediction pipeline cron schedule.
# Install with: crontab -e   (then paste the lines below — they assume the .venv at /opt/jarvis-app-1/.venv)

# 1. Live result fetch — every 5 min during matches.
*/5 * * * * /opt/jarvis-app-1/.venv/bin/python /opt/jarvis-app-1/scripts/wc2026_fetch_results.py >> /var/log/wc2026_fetch.log 2>&1

# 2. [RETIRED 2026-06-20] Old Elo+Poisson predictor as the Next-12 source.
#    Superseded by the three-spread ensemble engine (cron #10/#12), which now
#    owns wc2026_model_predictions.json. Left disabled for reference/rollback.
# 0 4 * * * /opt/jarvis-app-1/.venv/bin/python /opt/jarvis-app-1/scripts/wc2026_predictor.py >> /var/log/wc2026_predictor.log 2>&1

# 3. Lineup confirmation hook — hourly (script self-checks if any kickoff is within 90 min).
0 * * * * /opt/jarvis-app-1/.venv/bin/python /opt/jarvis-app-1/scripts/wc2026_lineup_hook.py >> /var/log/wc2026_lineup.log 2>&1

# 3b. Lineup-strength adjuster — every 30 min during the tournament. Uses the
#     confirmed XI from the hook + enriched player profiles to scale team lambdas.
#     Runs with single-threaded BLAS to keep stadium servers stable.
*/30 * * * * cd /opt/jarvis-app-1 && export OMP_NUM_THREADS=1 OPENBLAS_NUM_THREADS=1 MKL_NUM_THREADS=1 NUMEXPR_NUM_THREADS=1 && /opt/jarvis-app-1/.venv/bin/python /opt/jarvis-app-1/scripts/wc2026_lineup_adjuster.py --once >> /var/log/wc2026_lineup_adjuster.log 2>&1

# 3c. Vision match-video ingestion — every 4 hours. Processes any videos dropped
#     into server/data/vision_videos_in/ and extracts player/team tracking features.
0 */4 * * * /opt/jarvis-app-1/.venv/bin/python /opt/jarvis-app-1/scripts/wc2026_vision_ingest.py >> /var/log/wc2026_vision.log 2>&1

# 3d. Public web-data refresh — every 6 hours. Fetches openfootball + worldcup26.ir
#     results and merges them into fixtures + training history.
0 */6 * * * /opt/jarvis-app-1/.venv/bin/python /opt/jarvis-app-1/scripts/wc2026_web_data.py >> /var/log/wc2026_webdata.log 2>&1

# 4. Bookmaker odds scraper — every 6h. Refreshes value-bet flags.
0 */6 * * * /opt/jarvis-app-1/.venv/bin/python /opt/jarvis-app-1/scripts/wc2026_odds_scraper.py >> /var/log/wc2026_odds.log 2>&1

# 5. Player profiles refresh — weekly Sunday 03:00 UTC.
0 3 * * 0 OLLAMA_HOST=http://127.0.0.1:11434 /opt/jarvis-app-1/.venv/bin/python /opt/jarvis-app-1/scripts/wc2026_player_profiles.py >> /var/log/wc2026_profiles.log 2>&1

# 6. Historical data refresh — monthly on the 1st at 02:00 UTC.
0 2 1 * * /opt/jarvis-app-1/.venv/bin/python /opt/jarvis-app-1/scripts/wc2026_historical_data.py >> /var/log/wc2026_history.log 2>&1

# 7. Wasabi snapshot backup — hourly. Syncs server/data/wc2026_*.{json,csv,db} to s3://jarvis-wc2026/snapshots/<iso>/
0 * * * * /opt/jarvis-app-1/.venv/bin/python /opt/jarvis-app-1/scripts/wc2026_wasabi_backup.py >> /var/log/wc2026_wasabi.log 2>&1

# 8. T-8h prediction locker — every 15 min. For every fixture whose kickoff
#    is within the next 8 hours, freezes the current model prediction into
#    server/data/wc2026_predictions.db so it cannot be rewritten retroactively.
*/15 * * * * /opt/jarvis-app-1/.venv/bin/python /opt/jarvis-app-1/scripts/wc2026_lock_predictions.py >> /var/log/wc2026_lock.log 2>&1

# 9. Knockout bracket filler — every 4 hours during the tournament. Reads
#    group standings + completed knockout results from wc2026_results.json
#    and rewrites the 'Winner X' / 'Runner-up X' / 'Winner 73' placeholders
#    in wc2026_fixtures_all.json with concrete team names, then re-runs
#    wc2026_predictor.py so the new bracket gets predicted.
0 */4 * * * /opt/jarvis-app-1/.venv/bin/python /opt/jarvis-app-1/scripts/wc2026_knockout_filler.py >> /var/log/wc2026_filler.log 2>&1

# Notes:
# - All scripts are tolerant of network failures — they log and exit clean without crashing.
# - Dashboard polls /jarvis/data/wc2026_results.json every 60s, so updates surface within ~5-65 sec of cron firing.
# - To install: `crontab -l > /tmp/c; cat /opt/jarvis-app-1/scripts/wc2026_cron.txt >> /tmp/c; crontab /tmp/c`
# - To verify: `crontab -l | grep wc2026`

# 10. Three-spread prediction engine — daily backtest + slip regen (balanced).
#     04:15 UTC, after the predictor refresh. Re-selects model weights + rewrites
#     wc2026_backtest_results.json / wc2026_model_registry.json / wc2026_bet_slips.json.
15 4 * * * /opt/jarvis-app-1/.venv/bin/python /opt/jarvis-app-1/scripts/worldcup_prediction_engine.py --mode backtest --strategy balanced --runs 10000 >> /var/log/wc2026_engine.log 2>&1

# 11. Bet builder — regenerate three priced slips from live ESPN/DraftKings odds.
#     Every 2 hours during the tournament. Writes wc2026_bet_builder.json.
0 */2 * * * /opt/jarvis-app-1/.venv/bin/python /opt/jarvis-app-1/scripts/wc2026_bet_builder.py --stake 90 --legs 4 --strategy balanced --mode fixtures >> /var/log/wc2026_betbuilder.log 2>&1

# 12. Engine predict-upcoming — owns wc2026_model_predictions.json (the live
#     Next-12 source). Lighter than the full backtest; refreshes every 2 hours
#     offset 30 min from the bet builder. Replaces retired cron #2.
30 */2 * * * /opt/jarvis-app-1/.venv/bin/python /opt/jarvis-app-1/scripts/worldcup_prediction_engine.py --mode predict-upcoming --strategy balanced >> /var/log/wc2026_predictupcoming.log 2>&1

# 13. SBS watch-link cache refresh — hourly (resolver self-honours per-status TTLs).
0 * * * * /opt/jarvis-app-1/.venv/bin/python /opt/jarvis-app-1/scripts/sbs_watch_links.py >> /var/log/wc2026_sbs.log 2>&1

# 15. Build-manifest refresh — hourly, so the UI footer's "last sync" never goes
#     stale. Writes wc2026_BUILD_MANIFEST.json with a current generated_at.
5 * * * * /opt/jarvis-app-1/.venv/bin/python /opt/jarvis-app-1/scripts/wc2026_build_manifest.py >> /var/log/wc2026_manifest.log 2>&1

# 14. CONTINUOUS LEARNING — daily 04:45 UTC, after results settle. Trains a
#     recency-weighted walk-forward candidate, scores it OUT-OF-SAMPLE, and
#     promotes it ONLY if it beats the active model's objective (no calibration
#     drop, no coverage-cost blow-up). Then re-predicts + rebuilds slips. Writes
#     wc2026_model_registry.json + wc2026_active_model.json + model_versions/.
#     Single-threaded BLAS prevents numpy/scipy thread storms during weight learning.
#     The wrapper also enforces these env vars inside the Python process.
45 4 * * * /opt/jarvis-app-1/scripts/wc2026_daily_update.sh >> /var/log/wc2026_learning.log 2>&1
